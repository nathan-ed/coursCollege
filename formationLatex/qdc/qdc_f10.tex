\begin{frame}{\GetCompetence{Fonc21}}
Calculer l'aire du rectangle $ABCD$ avec $A(0;2)$, $B(-1;3)$, $C(1;5)$ et $D(2;4)$.
\onslide<2->{On calcule la longueur de $AB$ et de $BC$, puis on applique la formule de l'aire d'un rectangle.
	\begin{align*}
		\text{dist}(A;B) &= \sqrt{(-1-0)^2 + (3-2)^2} = \sqrt{1+1} = \sqrt{2},\\
		\text{dist}(B;C) &= \sqrt{(1+1)^2 + (5-3)^2} = \sqrt{4+4} = \sqrt{8},\\
		\text{Aire}(ABCD) &= AB \times BC = \sqrt{2} \times \sqrt{8} = \sqrt{16} = 4 \text{ unités au carré}.
\end{align*}}
