\begin{frame}{\GetCompetence{Fonc21}}
Déterminer l'aire du triangle $ABC$ où $A(1;-2)$, $B(-2;-3)$ et $C(6;0)$.
\end{frame}

\begin{frame}
	\begin{enumerate}
		\item Droite $BC$ est d'équation $y=\dfrac{3}{8}x-\dfrac{9}{4}$.
		\item La hauteur issue de $A$ est d'équation $y=-\dfrac{8}{3}x+\dfrac{2}{3}$.
		\item Le point d'intersection entre la hauteur et $BC$ est $H\left(\dfrac{70}{73};-\dfrac{138}{73}\right)$.
		\item Distance entre $A$ et $H$ : $\sqrt{\left(\dfrac{70}{73}-1\right)^2+\left(-\dfrac{138}{73}+2\right)^2}=\dfrac{\sqrt{73}}{73}$.
		\item Distance entre $B$ et $C$ : $\sqrt{(6+2)^2+(0+3)^2}=\sqrt{64+9}=\sqrt{73}$.
		\item  Donc $\text{Aire}(ABC)=\dfrac{1}{2}\cdot\sqrt{73}\cdot\dfrac{\sqrt{73}}{73}=\dfrac{1}{2}$ unités au carré.
	\end{enumerate}
\end{frame}
