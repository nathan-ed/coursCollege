\begin{frame}{\GetCompetence{Fonc22}}
	Déterminer le domaine de définition des fonctions suivantes.
	\begin{enumerate}
		\item $f(x) = \sqrt{x^2 - 4}$
		\item $g(x) = \dfrac{1}{x - 2}$
		\item $h(x) = \dfrac{1}{\sqrt{-x^2 + 2x + 3}}$
	\end{enumerate}
	\onslide<2->{\textbf{Solution :}
		\begin{enumerate}
		\item $x^2 - 4 \geq 0 \Rightarrow x^2 \geq 4 \Rightarrow x \geq 2 \text{ ou } x \leq -2$. Donc, $D_f = \left] -\infty, -2 \right] \cup \left[ 2, +\infty \right[$.
		\item $x - 2 \neq 0 \Rightarrow x \neq 2$. Donc, $D_g = \mathbb{R}\setminus\{2\}$.
		\item $-x^2 + 2x + 3 > 0 \Rightarrow x^2 - 2x - 3 < 0 \Rightarrow (x - 3)(x + 1) < 0$. Donc, $D_h = \left] -1, 3 \right[$.
\end{enumerate}}
\end{frame}
