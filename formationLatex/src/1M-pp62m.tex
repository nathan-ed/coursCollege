\contenu{
\tcblower
Soit la fonction $f$ définie sur $\mathbb{R}$ par 
\[f(x)=\begin{cases}\dfrac{x^3-8}{x-2}& \text{ si } x\neq 2,\\
c &\text{ si } x=2\end{cases}\]
pour $c\in \mathbb{R}$
\begin{tasks}
	\task Expliquez pourquoi cette fonction est continue sur $\mathbb{R}\setminus \{2\}$.
	\task Pour quelles valeurs de $c$ la fonction est-elle continue sur $\mathbb{R}$~?
\end{tasks}
}
\correction{
	\tcblower
\begin{tasks}
	\task Si on enlève $2$ de $D_f$, alors on peut simplifier l'expression de la fonction (division polynomiale) $f(x)=x^2+2x+4$ qui est une expression polynomiale, donc continue.
	\task $c=12$.
\end{tasks}
}
